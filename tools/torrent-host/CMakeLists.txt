# vantara-torrent: محرك libtorrent الأصلي لويندوز (ويبنى على لينكس للاختبار).
# ويندوز: vcpkg (libtorrent:x64-windows-static) فيخرج ملف تنفيذي واحد بلا DLL خارجية.
cmake_minimum_required(VERSION 3.21)
project(vantara_torrent CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(LibtorrentRasterbar CONFIG QUIET)
add_executable(vantara-torrent vantara-torrent.cpp)
if(LibtorrentRasterbar_FOUND)
  target_link_libraries(vantara-torrent PRIVATE LibtorrentRasterbar::torrent-rasterbar)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LT REQUIRED IMPORTED_TARGET libtorrent-rasterbar)
  target_link_libraries(vantara-torrent PRIVATE PkgConfig::LT)
endif()
find_package(Threads REQUIRED)
target_link_libraries(vantara-torrent PRIVATE Threads::Threads)
if(WIN32)
  target_compile_definitions(vantara-torrent PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(vantara-torrent PRIVATE ws2_32 iphlpapi)
endif()
if(NOT MSVC)
  target_compile_options(vantara-torrent PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
endif()
